cmake_minimum_required(VERSION 3.20)
project(EmbeddedCoreChapter11 VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
set(EMBEDDEDCORE_DEPENDENCY_MODE "PACKAGE" CACHE STRING "PACKAGE or FETCH")
set_property(CACHE EMBEDDEDCORE_DEPENDENCY_MODE PROPERTY STRINGS PACKAGE FETCH)
if(EMBEDDEDCORE_DEPENDENCY_MODE STREQUAL "PACKAGE")
  find_package(TelemetryCodec 1.2 CONFIG REQUIRED)
elseif(EMBEDDEDCORE_DEPENDENCY_MODE STREQUAL "FETCH")
  include(FetchContent)
  # SOURCE_DIR is intentionally local for this reproducible, network-free lab.
  FetchContent_Declare(TelemetryCodec
    SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/telemetrycodec")
  FetchContent_MakeAvailable(TelemetryCodec)
else()
  message(FATAL_ERROR "Unknown EMBEDDEDCORE_DEPENDENCY_MODE: ${EMBEDDEDCORE_DEPENDENCY_MODE}")
endif()
add_library(embeddedcore STATIC src/sensor.cpp src/telemetry.cpp)
add_library(EmbeddedCore::Core ALIAS embeddedcore)
target_compile_features(embeddedcore PUBLIC cxx_std_17)
target_include_directories(embeddedcore PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(embeddedcore PRIVATE TelemetryCodec::telemetrycodec)
add_executable(DeviceSimulator app/main.cpp)
target_link_libraries(DeviceSimulator PRIVATE EmbeddedCore::Core)
if(BUILD_TESTING)
  add_executable(telemetry_test tests/telemetry_test.cpp)
  target_link_libraries(telemetry_test PRIVATE EmbeddedCore::Core)
  add_test(NAME telemetry_contract COMMAND telemetry_test)
endif()
